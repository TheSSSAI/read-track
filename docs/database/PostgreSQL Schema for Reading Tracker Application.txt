-- Represents system users, their authentication details, and current status.
CREATE TABLE "User" (
    "userId" UUID PRIMARY KEY,
    "email" VARCHAR(255) NOT NULL CHECK ("email" ~* '^[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,}$'),
    "passwordHash" VARCHAR(255) NOT NULL,
    "subscriptionTier" VARCHAR(20) NOT NULL DEFAULT 'Free' CHECK ("subscriptionTier" IN ('Free', 'Premium')),
    "onboardingCompleted" BOOLEAN NOT NULL DEFAULT false,
    "isDeleted" BOOLEAN NOT NULL DEFAULT false,
    "createdAt" TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    "updatedAt" TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT "UC_User_Email" UNIQUE ("email")
);

-- Stores the history and status of user subscriptions.
CREATE TABLE "Subscription" (
    "subscriptionId" UUID PRIMARY KEY,
    "UserId" UUID NOT NULL,
    "status" VARCHAR(20) NOT NULL CHECK ("status" IN ('Active', 'Canceled', 'Expired')),
    "startDate" TIMESTAMPTZ NOT NULL,
    "endDate" TIMESTAMPTZ NOT NULL,
    "createdAt" TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    "updatedAt" TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);

-- A master table of all books known to the system.
CREATE TABLE "Book" (
    "bookId" UUID PRIMARY KEY,
    "googleBooksId" VARCHAR(50),
    "title" VARCHAR(255) NOT NULL,
    "author" VARCHAR(255) NOT NULL,
    "pageCount" INT CHECK ("pageCount" >= 0),
    "coverImageUrl" VARCHAR(512),
    "createdAt" TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    "updatedAt" TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT "UC_Book_GoogleBooksId" UNIQUE ("googleBooksId")
);

-- A join table representing a book in a user's personal library.
CREATE TABLE "UserBook" (
    "userBookId" UUID PRIMARY KEY,
    "UserId" UUID NOT NULL,
    "BookId" UUID NOT NULL,
    "bookTitle" VARCHAR(255) NOT NULL,
    "bookAuthor" VARCHAR(255) NOT NULL,
    "bookCoverImageUrl" VARCHAR(512),
    "shelf" VARCHAR(30) NOT NULL DEFAULT 'WantToRead' CHECK ("shelf" IN ('WantToRead', 'CurrentlyReading', 'Read')),
    "isReadOnly" BOOLEAN NOT NULL DEFAULT false,
    "addedAt" TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    "updatedAt" TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT "UC_UserBook_User_Book" UNIQUE ("UserId", "BookId")
);

-- Logs individual reading sessions for a book in a user's library.
CREATE TABLE "ReadingSession" (
    "readingSessionId" UUID PRIMARY KEY,
    "UserBookId" UUID NOT NULL,
    "UserId" UUID NOT NULL,
    "startTime" TIMESTAMPTZ NOT NULL,
    "endTime" TIMESTAMPTZ NOT NULL,
    "pagesRead" INT CHECK ("pagesRead" >= 0),
    "notes" TEXT,
    "createdAt" TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    "updatedAt" TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
) PARTITION BY RANGE ("startTime");

-- Stores user-defined reading goals and tracks their progress.
CREATE TABLE "Goal" (
    "goalId" UUID PRIMARY KEY,
    "UserId" UUID NOT NULL,
    "goalType" VARCHAR(20) NOT NULL CHECK ("goalType" IN ('Books', 'Pages', 'Time')),
    "period" VARCHAR(20) NOT NULL CHECK ("period" IN ('Day', 'Week', 'Month', 'Year')),
    "targetValue" INT NOT NULL CHECK ("targetValue" > 0),
    "currentValue" INT NOT NULL DEFAULT 0 CHECK ("currentValue" >= 0),
    "startDate" TIMESTAMPTZ NOT NULL,
    "endDate" TIMESTAMPTZ NOT NULL,
    "isActive" BOOLEAN NOT NULL DEFAULT true,
    "createdAt" TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    "updatedAt" TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT "UC_Goal_User_Active_Type_Period_Start" UNIQUE ("UserId", "goalType", "period", "startDate", "isActive")
);

-- Stores AI-generated book recommendations and user feedback on them.
CREATE TABLE "Recommendation" (
    "recommendationId" UUID PRIMARY KEY,
    "UserId" UUID NOT NULL,
    "recommendedBookTitle" VARCHAR(255) NOT NULL,
    "recommendedBookAuthor" VARCHAR(255) NOT NULL,
    "reasoning" TEXT NOT NULL,
    "feedback" VARCHAR(20) CHECK ("feedback" IN ('Like', 'Dislike')),
    "generatedAt" TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);

-- Stores words and definitions saved by the user.
CREATE TABLE "VocabularyItem" (
    "vocabularyItemId" UUID PRIMARY KEY,
    "UserId" UUID NOT NULL,

    "definition" TEXT,

-- Tracks the status of user-initiated data export requests.
CREATE TABLE "DataExportJob" (
    "dataExportJobId" UUID PRIMARY KEY,

    "fileUrl" VARCHAR(512),

    "updatedAt" TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP

-- A key-value store for user-specific application settings like theme.
CREATE TABLE "UserSetting" (

    "settingValue" VARCHAR(255) NOT NULL,

-- A summary table for daily user reading statistics.

    "UserId" UUID NOT NULL,
